2 Firestore Security Rules (Copy & Paste in Firebase)

Security update: Firestore Database → Rules tab mein ye complete code paste karke Publish karein. Catalog public-read hai; orders/customer records private hain. Admin access Firebase Authentication se hi hoga.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isAdmin() {
      return request.auth != null && (
        request.auth.token.email == 'sdhakar592@gmail.com' ||
        request.auth.token.email == 'dhakadgsm592@gmail.com'
      );
    }
    // Public catalog: visitors can read; only authenticated admins can change it.
    match /downloads/{docId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }
    match /toolrents/{docId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }
    match /services/{docId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }
    match /announcements/{docId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }
    // Optional Order Counter for sequential #DGSM-0001 IDs
    match /orderCounters/{counterId} {
      allow read, create, update: if request.auth != null;
      allow delete: if isAdmin();
    }
    // Private customer orders (Matches #DGSM-0001 format)
    match /orders/{orderId} {
      allow read: if isAdmin() || (request.auth != null && resource.data.customerUid == request.auth.uid);
      allow create: if request.auth != null
        && request.resource.data.customerUid == request.auth.uid
        && request.resource.data.customerEmail == request.auth.token.email
        && request.resource.data.customerName is string
        && request.resource.data.customerName.size() > 0
        && request.resource.data.customerName.size() <= 100
        && request.resource.data.customerPhone is string
        && request.resource.data.customerPhone.size() <= 20
        && request.resource.data.itemTitle is string
        && request.resource.data.itemTitle.size() > 0
        && request.resource.data.itemTitle.size() <= 200
        && request.resource.data.itemPrice is string
        && request.resource.data.status == 'Pending'
        && request.resource.data.orderCode is string
        && request.resource.data.orderCode.matches('^#?DGSM-[0-9]+$')
        && request.resource.data.createdAt is int;
      allow update: if isAdmin() || (request.auth != null
        && resource.data.customerUid == request.auth.uid
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['customerReply', 'customerReplyAt', 'updatedAt']));
      allow delete: if isAdmin();
    }
    match /customers/{userId} {
      allow read: if isAdmin() || (request.auth != null && request.auth.uid == userId);
      allow create, update: if request.auth != null && request.auth.uid == userId
        && request.resource.data.email == request.auth.token.email;
      allow delete: if isAdmin();
    }
    // Deny every collection not explicitly listed above.
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
